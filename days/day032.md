-- 1.   MySQL은 _데이터베이스 관리 시스템(DBMS)이다.
-- 2.  (#1 _데이터)는 수집된 사실이나 값을 의미하고 
       (#2 _정보)는 의미를 제공하는 데이터를 의미한다.
-- 3. 대표적인 데이터베이스 종류에는 _oracle, _mysql, _mssql 등이 있다.
-- 4.  RDBMS는 _relational Database Management System의 약자이다.
-- 5. RDBMS의 구성요소는   _개체(entity,table), _관계(relationship), _속성(attribut : 필드) 
-- 6. 데이터베이스 언어
-- DDL(  정의어   )  CREATE, ALTER, DROP    -> cad 
-- DML(  제어어   )    _insert, _select, _update , _delete   ->crud
-- DCL(  제어어   )     _grant, _revoke

-- 7. 데이터베이스  mbasic, 를 생성하는 SQL 명령어를 작성하시오.
        create database mbasic;

-- 8.  생성된 데이터베이스 목록을 확인하는 SQL 명령어를 작성하시오.
        show databases;

-- 9. 데이터베이스 db703을 삭제하는 SQL 명령어를 작성하시오.
        drop database db703;
